cmake_minimum_required(VERSION 3.16)
project(DGSourceFreeDataset LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Boost REQUIRED)
add_executable(dg_dataset src/dataset_driver.cpp src/cole_cole.cpp)
target_include_directories(dg_dataset PRIVATE include ${Boost_INCLUDE_DIRS})
target_compile_options(dg_dataset PRIVATE -O3 -Wall -Wextra)
if(APPLE AND EXISTS "/opt/homebrew/opt/libomp/include/omp.h")
  target_include_directories(dg_dataset PRIVATE /opt/homebrew/opt/libomp/include)
  target_compile_options(dg_dataset PRIVATE -Xpreprocessor -fopenmp)
  target_compile_definitions(dg_dataset PRIVATE COLECOLE_ENABLE_OPENMP=1)
  target_link_libraries(dg_dataset PRIVATE /opt/homebrew/opt/libomp/lib/libomp.dylib)
  target_link_options(dg_dataset PRIVATE -Wl,-rpath,/opt/homebrew/opt/libomp/lib)
endif()
